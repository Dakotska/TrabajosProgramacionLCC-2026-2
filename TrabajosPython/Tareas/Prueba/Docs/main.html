 28def InicioJuego() -> None:
 29    """
 30        Inicializa la partida repartiendo dos cartas de manera alternada al crupier y al jugador.
 31        Limpia la mesa, restablece las variables globales y reparte las manos iniciales.
 32        Posteriormente, verifica si con la mano inicial hay algún Blackjack y decide al ganador; de lo contrario, sigue el juego normal.
 33
 34        Globals:
 35            PALOS (list): Palos predefinidos de las cartas.
 36            DENOMINACIONES (list): Denominaciones predefinidas de las cartas.
 37            baraja (list): Baraja completa de 52 combinaciones según los palos y denominaciones.
 38            crupier (list): Mano del crupier.
 39            segundaCartaCrupier (list): Segunda carta del crupier, la cual está oculta.
 40            puntCrupier (int): Puntuación total actual del crupier.
 41            jugador (list): Mano del jugador.
 42            puntJugador (int): Puntuación total actual del jugador.
 43    """
 44    global PALOS, DENOMINACIONES, baraja, crupier, segundaCartaCrupier, puntCrupier, jugador, puntJugador
 45
 46    # Se acomodan los botones en la interfaz y se habilitan para el usuario.
 47    menu.place_forget()
 48    textoMenu.grid_forget()
 49    botonInicio.grid_forget()
 50    botonSalir.grid_forget()
 51    botonPedir.config(state="normal")
 52    botonPlantar.config(state="normal")
 53    for dibujo in marcoManoCrupier.winfo_children():
 54        dibujo.destroy()
 55    for dibujo in marcoManoJugador.winfo_children():
 56        dibujo.destroy()
 57    barraPuntJugador.config(bg="green")
 58
 59    # Se restablecen las variables globales para la nueva partida y se baraja el mazo.
 60    baraja = [(den, palo) for palo in PALOS for den in DENOMINACIONES]
 61    crupier = []
 62    segundaCartaCrupier = []
 63    puntCrupier = 0
 64    jugador = []
 65    puntJugador = 0
 66
 67    shuffle(baraja)
 68
 69    # Se reparten cuatro cartas de forma alternada: dos para el jugador y dos para el crupier.
 70    for i in range(2):
 71        JalarCarta(True)
 72        JalarCarta(False)
 73
 74    # Se verifica si algún participante obtuvo Blackjack de manera natural.
 75    # Si es así, se muestran las cartas del crupier, se determina el ganador y se finaliza la partida.
 76    estado = VerificarBlackjack()
 77    if estado != 0:
 78        hijos = marcoManoCrupier.winfo_children()
 79        hijos[1].destroy()
 80        MostrarCarta(segundaCartaCrupier[0], False)
 81        textoPuntCrupier.configure(text=f"{puntCrupier}/21")
 82        bordeBarraCrupier.columnconfigure(0, weight=puntCrupier)
 83        bordeBarraCrupier.columnconfigure(1, weight=(21 - puntCrupier))
 84        barraPuntCrupier.configure(bg="green")
 85        if estado == 1:
 86            MostrarFinJuego(1)
 87        elif estado == 2:
 88            MostrarFinJuego(2)
 89        elif estado == 3:
 90            MostrarFinJuego(3)
 91

 96def Pedir():
 97    """
 98        Función ejecutada por el botón "Pedir".
 99        Le entrega una carta adicional al jugador y evalúa si este excedió los 21 puntos.

113        barraPuntCrupier.configure(bg="green")
114        MostrarFinJuego(1)

123def Plantarse() -> None:
124    """
125        Función ejecutada por el botón "Plantar".
126        Da por terminado el turno del jugador. Revela la carta oculta del crupier y
127        hace que este pida cartas obligatoriamente hasta alcanzar al menos 17 puntos.
128        Posteriormente, verifica los puntajes finales y decide al ganador.
129    """
130
131    # Se deshabilitan los botones de acción ("Pedir" y "Plantar").
132    botonPedir.config(state="disabled")
133    botonPlantar.config(state="disabled")
134
135    # Se muestra la segunda carta del crupier en la interfaz.
136    hijos = marcoManoCrupier.winfo_children()
137    hijos[1].destroy()
138    MostrarCarta(segundaCartaCrupier[0], False)
139
140    # El crupier toma cartas adicionales de la baraja mientras su puntuación total sea menor a 17 puntos.
141    if puntCrupier >= 17:
142        textoPuntCrupier.configure(text=f"{puntCrupier}/21")
143        bordeBarraCrupier.columnconfigure(0, weight=puntCrupier)
144        bordeBarraCrupier.columnconfigure(1, weight=(21 - puntCrupier))
145        barraPuntCrupier.configure(bg="green")
146    while puntCrupier < 17:
147        JalarCarta(False)
148
149    # Se verifica si el crupier excede los 21 puntos. Posteriormente, se evalúan los puntajes finales
150    # para determinar si hubo empate, victoria del jugador o victoria del crupier.
151    if VerificarSobrepuntaje() == 1:
152        MostrarFinJuego(2)
153    estado = VerificarPuntajesFinales()
154    if estado == 3:
155        MostrarFinJuego(3)
156    elif estado == 1:
157        MostrarFinJuego(1)
158    elif estado == 2:
159        MostrarFinJuego(2)

165def JalarCarta(esJugador: bool):
166    """
167        Agrega una carta de la baraja a la mano del jugador o del crupier.
168
169        Actualiza las variables globales correspondientes (`jugador` o `crupier`),
170        ajusta el puntaje y renderiza la barra de progreso en la interfaz.
171
172        Parameters:
173            esJugador (bool): `True` si el que pide la carta es el jugador, `False` si es el crupier.
174
175        Raises:
176            TypeError: Si el argumento `esJugador` no es un valor booleano.
177
178        Examples:
179            >>> JalarCarta(True)
180            >>> JalarCarta(False)
181    """
182    # Manejo de excepciones para validar el tipo de dato.
183    if not isinstance(esJugador, bool):
184        raise TypeError("ERROR: El argumento 'esJugador' no es un valor booleano.")
185
186    # Definición de variables globales.
187    global crupier, jugador, puntCrupier, puntJugador
188
189    # Evalúa si la carta solicitada es para el jugador o para el crupier.
190    if esJugador:
191        # Se extrae una carta de la baraja, se agrega a la mano del jugador y se renderiza en la interfaz.
192        carta = baraja.pop()
193        jugador.append(carta)
194        MostrarCarta(carta, True)
195
196        # Si la carta es un As, Jota, Reina o Rey, se llama a la función ArreglarValores para ajustar
197        # su valor antes de sumarlo a la puntuación del jugador.
198        if carta[0] in "AJQK":
199            jugador.pop()
200            jugador.append(ArreglarValores(True, carta))
201
202        puntJugador += int(jugador[-1][0])
203        textoPuntJugador.configure(text=f"{puntJugador}/21")
204
205        # Se ajusta el color y progreso de la barra de puntuación dependiendo de si el jugador superó los 21 puntos.
206        if puntJugador <= 21:
207            bordeBarraJugador.columnconfigure(0, weight=puntJugador)
208            bordeBarraJugador.columnconfigure(1, weight=(21 - puntJugador))
209        else:
210            bordeBarraJugador.columnconfigure(0, weight=21)
211            bordeBarraJugador.columnconfigure(1, weight=0)
212            barraPuntJugador.configure(bg="red")
213    else:
214        # Se extrae una carta de la baraja, se agrega a la mano del crupier y se renderiza en la interfaz.
215        carta = baraja.pop()
216        crupier.append(carta)
217        MostrarCarta(carta, False)
218
219        # Si la carta es un As, Jota, Reina o Rey, se llama a la función ArreglarValores para ajustar
220        # su valor antes de sumarlo a la puntuación del crupier.
221        if carta[0] in "AJQK":
222            crupier.pop()
223            crupier.append(ArreglarValores(False, carta))
224
225        puntCrupier += int(crupier[-1][0])
226
227        # Si el jugador aún no se ha plantado, se oculta la puntuación total del crupier.
228        # De lo contrario, se muestra su puntaje actual y se actualiza el estado de su barra de puntuación.
229        if len(crupier) <= 2:
230            textoPuntCrupier.configure(text=f"?/21")
231            puntCrupierConocida = int(crupier[0][0])
232            bordeBarraCrupier.columnconfigure(0, weight=puntCrupierConocida)
233            bordeBarraCrupier.columnconfigure(1, weight=(21 - puntCrupierConocida))
234            barraPuntCrupier.configure(bg="yellow")
235        else:
236            textoPuntCrupier.configure(text=f"{puntCrupier}/21")
237            if puntCrupier <= 21:
238                bordeBarraCrupier.columnconfigure(0, weight=puntCrupier)
239                bordeBarraCrupier.columnconfigure(1, weight=(21 - puntCrupier))
240                barraPuntCrupier.configure(bg="green")
241            else:
242                bordeBarraCrupier.columnconfigure(0, weight=21)
243                bordeBarraCrupier.columnconfigure(1, weight=0)
244                barraPuntCrupier.configure(bg="red")
245

250def VerificarBlackjack() -> int:
251    """
252        Evalúa si el puntaje del crupier o del jugador es exactamente de 21 puntos.
253
254        Returns:
255            int: Devuelve `0` si nadie tiene 21 (el juego sigue), `1` si el crupier tiene 21,
256            `2` si el jugador tiene 21, y `3` si ambos tienen 21 (empate).
257
258        Examples:
259            >>> puntJugador = 21
260            >>> puntCrupier = 17
261            >>> VerificarBlackjack()
262            2
263    """
264    # Definición de variables globales.
265    global puntJugador, puntCrupier
266
267    # Se evalúa si hubo un empate por Blackjack natural, o victoria de alguna de las partes por la misma vía.
268    # Si no ocurre ninguno de estos escenarios, se devuelve 0.
269    if puntCrupier == 21 and puntJugador == 21:
270        return 3
271    elif puntJugador == 21:
272        return 2
273    elif puntCrupier == 21:
274        return 1
275    else:
276        return 0

282def VerificarSobrepuntaje() -> int:
283    """
284        Evalúa si el puntaje del crupier o del jugador superó los 21 puntos.
285
286        Returns:
287            int: Devuelve `0` si nadie se ha pasado, `1` si el crupier se pasó de 21,
288            `2` si el jugador se pasó de 21.
289    """
290    # Definición de variables globales.
291    global puntJugador, puntCrupier
292
293    # Evalúa si el crupier o el jugador excedieron los 21 puntos. En caso contrario, se devuelve 0.
294    if puntCrupier > 21:
295        return 1
296    elif puntJugador > 21:
297        return 2
298    return 0

304def VerificarPuntajesFinales() -> int:
305    """
306        Evalúa el puntaje final del crupier y del jugador para determinar al ganador.
307
308        Esta función asume que ninguno de los participantes se ha pasado de los 21 puntos.
309
310        Returns:
311            int: Devuelve `1` si el crupier ganó, `2` si el jugador ganó,
312            y `3` si fue un empate.
313    """
314    # Definición de variables globales.
315    global puntJugador, puntCrupier
316
317    # Evalúa los puntajes finales válidos (menores o iguales a 21) para determinar al ganador o un posible empate.
318    # Si no se cumple ninguna condición, se devuelve 0.
319    if (puntJugador == puntCrupier) and (puntJugador <= 21 and puntCrupier <= 21):
320        return 3
321    elif (puntJugador > puntCrupier) and puntJugador <= 21:
322        return 2
323    elif (puntCrupier > puntJugador) and puntCrupier <= 21:
324        return 1
325    else:
326        return 0

332def ArreglarValores(esJugador: bool, carta: tuple) -> tuple:
333    """
334        Asigna el valor numérico correspondiente a una carta especial ("A", "J", "Q", "K").
335
336        Si la carta es un As ("A"), evalúa si su valor debe ser 1 u 11, dependiendo de si
337        la puntuación total superaría los 21 puntos.
338
339        Parameters:
340            esJugador (bool): `True` si se evalúa la carta del jugador, `False` si es del crupier.
341            carta (tuple): Tupla que contiene la carta a evaluar.
342
343        Returns:
344            tuple: Nueva tupla con el valor numérico ajustado y el mismo palo.
345    """
346    # Evalúa si se está ajustando el valor para el jugador o para el crupier.
347    if esJugador:
348        # Si la carta es Jota, Reina o Rey, devuelve un valor numérico de 10.
349        if carta[0] in "JQK":
350            return (10,carta[1])
351        # Si es un As, devuelve el valor que más le conviene al jugador (1 u 11) para evitar superar los 21 puntos.
352        else:
353            if 11 + puntJugador > 21:
354                return (1,carta[1])
355            else:
356                return (11,carta[1])
357    else:
358        # Si la carta es Jota, Reina o Rey, devuelve un valor numérico de 10.
359        if carta[0] in "JQK":
360            return (10,carta[1])
361        # Si es un As, devuelve el valor que más le conviene al crupier (1 u 11) para evitar superar los 21 puntos.
362        else:
363            if 11 + puntCrupier > 21:
364                return (1,carta[1])
365            else:
366                return (11,carta[1])

372def MostrarCarta(carta: tuple, esJugador: bool):
373    """
374        Genera y coloca gráficamente una carta en la interfaz (Tkinter).
375
376        Si es la segunda carta del crupier en el turno inicial, se imprime boca abajo
377        (oculta) hasta que sea el turno de revelarla.
378
379        Parameters:
380            carta (tuple): Tupla que representa la carta (valor, palo).
381            esJugador (bool): `True` si la carta pertenece al jugador, `False` al crupier.
382
383        Raises:
384            TypeError: Si `carta` no es una tupla o `esJugador` no es un valor booleano.
385            ValueError: Si el formato de la carta es inválido.
386    """
387    # Manejo de excepciones para validar los tipos de datos de entrada.
388    if not isinstance(carta, tuple) or not isinstance(esJugador, bool):
389        raise TypeError("El argumento 'carta' debe ser una tupla y el argumento 'esJugador' debe ser un valor booleano.")
390    if not len(carta) == 2:
391        raise ValueError("El argumento 'carta' debe tener 2 elementos dentro de la tupla.")
392    if not 1 <= len(carta[0]) <= 2 or not carta[1] in PALOS:
393        raise ValueError("El primer elemento del argumento 'carta' debe tener 1 o 2 caracteres, y el segundo elemento debe estar en la constante PALOS.")
394    if len(carta[0]) == 2 and not carta[0] == "10":
395        raise ValueError("El primer elemento del argumento 'carta' que tiene 2 caracteres debe ser la cadena '10'.")
396    
397    # Definición de variables globales.
398    global crupier, jugador, segundaCartaCrupier
399
400    # Asignación del valor y palo de la carta a variables locales.
401    valor = carta[0]
402    palo = carta[1]
403
404    # Renderizado adaptado para cartas cuyo valor es de un solo carácter.
405    if len(valor) == 1:
406        textoCarta = (
407            f"┌───────┐\n"
408            f"│{valor}      │\n"
409            f"│       │\n"
410            f"│   {palo}   │\n"
411            f"│       │\n"
412            f"│      {valor}│\n"
413            f"└───────┘"
414        )
415    # Renderizado adaptado para cartas cuyo valor es de dos caracteres.
416    else:
417        textoCarta = (
418            f"┌───────┐\n"
419            f"│{valor}     │\n"
420            f"│       │\n"
421            f"│   {palo}   │\n"
422            f"│       │\n"
423            f"│     {valor}│\n"
424            f"└───────┘"
425        )
426
427    # Evalúa si la carta a mostrar pertenece al jugador o al crupier.
428    if esJugador:
429        # Renderiza y configura la carta del jugador de forma visible.
430        dibujoCarta = tk.Label(marcoManoJugador, text=textoCarta, font=("Courier New", 11, "bold"), bg="white")
431        dibujoCarta.grid(column=jugador.index(carta), row=0, sticky="w")
432    else:
433        # Determina si es la primera o la segunda carta del crupier.
434        # La primera se muestra normalmente, mientras que la segunda se oculta temporalmente.
435        if len(segundaCartaCrupier) != 0 and carta == segundaCartaCrupier[0]:
436            dibujoCarta = tk.Label(marcoManoCrupier, text=textoCarta, font=("Courier New", 11, "bold"), bg="white")
437            dibujoCarta.grid(column=4, row=0, sticky="e")
438        else:
439            if len(crupier) == 2 and crupier.index(carta) == 1:
440                textoCarta = (
441                    f"┌───────┐\n"
442                    f"│ ▓▓▓▓▓ │\n"
443                    f"│ ▓▓▓▓▓ │\n"
444                    f"│ ▓▓▓▓▓ │\n"
445                    f"│ ▓▓▓▓▓ │\n"
446                    f"│ ▓▓▓▓▓ │\n"
447                    f"└───────┘"
448                )
449                segundaCartaCrupier.append(carta)
450
451            dibujoCarta = tk.Label(marcoManoCrupier, text=textoCarta, font=("Courier New", 11, "bold"), bg="white")
452            if len(segundaCartaCrupier) != 0 and carta in segundaCartaCrupier:
453                dibujoCarta.grid(column=4, row=0, sticky="e")
454            else:
455                columna_pos = 5 - (len(crupier) - 1)
456                dibujoCarta.grid(column=columna_pos, row=0, sticky="e")
457

462def MostrarFinJuego(estado: int):
463    """
464        Muestra la interfaz de final de partida.
465
466        Despliega un menú superpuesto anunciando al ganador, desactiva los botones
467        de acción y muestra las opciones de "REINICIAR" o "SALIR".
468
469        Parameters:
470            estado (int): Representa el resultado (1 = Gana crupier, 2 = Gana jugador, 3 = Empate).
471
472        Raises:
473            TypeError: Si el `estado` proporcionado no es un número entero.
474            ValueError: Si el `estado` no está en el rango válido (1, 2, 3).
475    """
476    # Manejo de excepciones para validar que el estado sea válido.
477    if not isinstance(estado, int):
478        raise TypeError("El argumento 'estado' debe ser un número entero.")
479    if not 1 <= estado <= 3:
480        raise ValueError("El argumento 'estado' debe ser un número entero de 1 a 3.")
481
482    # Se deshabilitan los botones de acción y se despliega el menú de fin de juego.
483    botonPedir.config(state="disabled")
484    botonPlantar.config(state="disabled")
485
486    menu.place(width=app.winfo_width(), height=180, y=(app.winfo_height()-180))
487    menu.rowconfigure((0, 1), weight=1)
488    menu.columnconfigure(0, weight=1)
489
490    # Evalúa el estado final para determinar si ganó el crupier, ganó el jugador o hubo empate.
491    if estado == 1:
492        textoEstado = "Gana CRUPIER!"
493    elif estado == 2:
494        textoEstado = "Gana JUGADOR!"
495    else:
496        textoEstado = "EMPATE!"
